\makeatletter
\newcounter{observation}
\newenvironment{observation}[1]{
    \let\@makefntext\observation@makefntext
    \refstepcounter{observation}
    \fontfamily{cmss}\selectfont
    \begin{tcolorbox}[
        before upper={
            \setlength{\parskip}{0.15cm}
            \setlength{\parindent}{0pt}},
            enhanced jigsaw,
            breakable,
            colupper=observationcolor, 
            colback=white,
            colframe=observationcolorframe,
            title=Observación \thesection.\theobservation: #1]
    \savenotes
}{

    \endsavenotes
    \end{tcolorbox}
}
\makeatother
